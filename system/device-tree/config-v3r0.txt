auto_initramfs=1

# Enable DRM VC4 V3D driver

# max_framebuffers=2

# Don't have the firmware create an initial video= setting in cmdline.txt.
# Use the kernel's default instead.
disable_fw_kms_setup=1

# Disable compensation for displays with overscan
disable_overscan=1

# Run as fast as firmware / board allows
arm_boost=1

# disable bluetooth to use uart
dtoverlay=disable-bt

# enable uart tx / rx for iridigum
dtoverlay=uart1,txd1_pin=32,rxd1_pin=33

# Enable PPS for the GPS
dtoverlay=pps-gpio,gpio=13

# System RTC
dtoverlay=i2c-rtc,mcp7941x

# reset usb
gpio=24=op,dh
# start the GPS
gpio=6=op,dl
# IO Expander (SX1509B enable)
gpio=39=op,dh

# Front-panel buttons (kernel gpio-keys -> evdev). Status LEDs are on the SX1509B
# (driven by ctt-leds over I2C), so there is no gpio-led overlay on V3.
dtoverlay=gpio-key,gpio=17,active_low=0,gpio_pull=down,keycode=103,label=ctt-btn-up
dtoverlay=gpio-key,gpio=22,active_low=0,gpio_pull=down,keycode=108,label=ctt-btn-down
dtoverlay=gpio-key,gpio=27,active_low=0,gpio_pull=down,keycode=28,label=ctt-btn-select
dtoverlay=gpio-key,gpio=8,active_low=0,gpio_pull=down,keycode=1,label=ctt-btn-back

# BCM2711 USB filter. [pi4] matches EVERY 2711 variant (CM4, CM4S, Pi 4B); do not
# narrow it to [cm4], which does not match a CM4S. When it does not match, otg_mode
# is never applied and the module falls back to the legacy DWC2 controller — seen on
# a CM4S as "using dwc2" in the kernel log where a CM3 logs "dwc_otg", with unstable
# USB host behaviour on a fully populated hub.
[pi4]
# Enable host mode on the 2711 built-in XHCI USB controller.
# This line should be removed if the legacy DWC2 controller is required
# (e.g. for USB device mode) or if USB support is not required.
otg_mode=1

[all]
